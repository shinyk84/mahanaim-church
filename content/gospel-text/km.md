# តើអ្នកពិតជាមានសេចក្ដីសុខសាន្ត និងសុភមង្គលមែនដែរឬទេ?

នៅក្នុងជីវិតរស់នៅរបស់យើង ជារឿយៗយើងតែងតែជួបប្រទះនឹងបញ្ហាដែលមិនអាចដោះស្រាយបាន។ បញ្ហាគ្រួសារ បញ្ហាការងារ បញ្ហាកូនចៅ អនាគតដែលមិនច្បាស់លាស់ បញ្ហាខាងវិញ្ញាណ និងខាងរូបកាយ តែងតែស្ថិតនៅចំពោះមុខយើងជានិច្ច។ ដើម្បីដោះខ្លួនចេញពីបញ្ហាទាំងនេះ មនុស្សខ្លះធ្លាក់ចុះទៅក្នុងគ្រឿងញៀន ស្រា ឬរស់នៅក្នុងជីវិតដែលពាលាអាវ៉ាសែ។ មនុស្សខ្លះទៀត ដោយសារតែមិនមានចម្លើយ ក៏ពឹងផ្អែកលើសាសនា ជំនឿអរូបីយ៍ អំពើអាបធ្មប់ លុយកាក់ និងអំណាច ដើម្បីរស់នៅ។ ហេតុអ្វីបានជាបញ្ហាទាំងនេះកើតមានឡើង? តើដំណោះស្រាយដ៏ពិតប្រាកដគឺជាអ្វី?

## 1. មនុស្សកាលពីដើមដំបូង

ក្នុងចំណោមអ្វីៗទាំងអស់ដែលព្រះបានបង្កើត មានតែមនុស្សប៉ុណ្ណោះដែលត្រូវបានបង្កើតមកតាមរូបអង្គព្រះជាម្ចាស់។ ដូចជាត្រីត្រូវរស់នៅក្នុងទឹក បក្សីត្រូវរស់នៅលើអាកាស ហើយដើមឈើត្រូវចាក់ឫសនៅក្នុងដី មនុស្សដែលត្រូវបានបង្កើតមកតាមរូបអង្គព្រះជាម្ចាស់ ក៏ត្រូវរស់នៅជាមួយព្រះជាម្ចាស់ដូច្នោះដែរ។ [[GEN 1:27-28]]

## 2. អំពើបាបរបស់មនុស្ស

ប៉ុន្តែ មនុស្សត្រូវបានសាថាន់ល្បួងបញ្ឆោត ហើយបានប្រព្រឹត្តអំពើបាបដោយការមិនជឿព្រះ នៅទីបំផុតក៏បានដាច់ចេញពីព្រះជាម្ចាស់។ [[GEN 3:1-6]]
សាថាន់ប្រឆាំងទាស់នឹងព្រះជាម្ចាស់ នាំមកនូវបណ្ដាសានៃសេចក្ដីរងទុក្ខឈឺចាប់ ហើយនៅទីបំផុតធ្វើឲ្យមនុស្សវិនាសអន្តរាយ។ [[REV 12:9]] [[ROM 3:23]] [[ROM 6:23]]

## 3. បញ្ហាឬសគល់ដើមរបស់មនុស្ស ដែលបានដាច់ចេញពីព្រះ

(1) កូនរបស់អារក្ស៖ មនុស្សដែលដាច់ចេញពីព្រះជាម្ចាស់ គឺស្ថិតក្នុងស្ថានភាពស្លាប់ខាងវិញ្ញាណ ហើយត្រូវស្ថិតនៅក្រោមការគ្រប់គ្រងរបស់អារក្ស។ [[JOH 8:44]]

(2) ការថ្វាយបង្គំរូបព្រះក្លែងក្លាយ៖ ជាលទ្ធផល ពួកគេធ្លាក់ចុះក្នុងរូបព្រះក្លែងក្លាយនៃអបីយ៍ជំនឿ ដោយព្យាយាមមើលគ្រូទាយ ធ្វើពិធីរំដោះគ្រោះ ឬធ្វើអំពើល្អ ប៉ុន្តែនៅតែគ្មានសុភមង្គល ហើយត្រូវរស់នៅជាប់ចំណងនឹងជោគវាសនា ហោរាសាស្ត្រ និងឆ្នាំខែ ដែលមិនអាចជ្រើសរើសថ្ងៃ ខែ ទិស ឬកន្លែងបញ្ចុះសពតាមចិត្តបានឡើយ។ [[EXO 20:4-5]]

(3) ការរងទុក្ខឈឺចាប់ខាងសតិស្មារតី៖ ត្រូវជាប់ក្នុងបណ្តាសាដ៏ចម្លែក ដូចជាការព្រួយបារម្ភ ការត្អូញត្អែរ ភាពទទេ ជំងឺផ្លូវចិត្ត ជំងឺសរសៃប្រសាទ គេងមិនលក់ និងជំងឺបាក់ទឹកចិត្តជាដើម។ ទោះបីជាមានប្រវត្តិគ្រួសារល្អ មានចំណេះដឹងច្រើន និងមានលុយច្រើនក៏ដោយ ក៏ជីវិតនៅតែធ្លាក់ចុះទៅៗ។ នេះគឺជាស្ថានភាពដែលត្រូវទទួលសេចក្តីក្រោធ ពីកំណើត។ [[EPH 2:3]] [[MAT 11:28]]

(4) ការរងទុក្ខឈឺចាប់ខាងសាច់ឈាម៖ រងទុក្ខដោយជំងឺព្យាបាលមិនជា គ្រោះភ័យ ជំងឺតម្កាត់ សុបិនអាក្រក់ ខ្មោចសង្កត់ និងឈឺចាប់ពេញរាងកាយដោយរកមូលហេតុជំងឺមិនឃើញ។ [[ACT 8:4-8]]

(5) សេចក្តីស្លាប់ និងការជំនុំជម្រះទៅស្ថាននរក៖ បដិសេធព្រះជាម្ចាស់ ហើយឈ្លក់វង្វេងតែនឹងជីវិតបច្ចុប្បន្ន លុះដល់ទីបំផុតក៏ស្លាប់ទៅ ហើយត្រូវទទួលការជំនុំជម្រះអស់កល្បជានិច្ចទៅក្នុងស្ថាននរក។ [[LUK 16:19-31]] [[HEB 9:27]]

(6) មរតកខាងវិញ្ញាណ៖ នៅទីបំផុត បញ្ហាទាំងអស់ត្រូវបានផ្ទេរតទៅកូនចៅជំនាន់ក្រោយទៀត។ [[1CO 10:20]] [[MAT 27:25]]

## 4. ដំណោះស្រាយតែមួយគត់ - ព្រះយេស៊ូវគ្រីស្ទ

អ្នកណាក៏ដោយដែលជឿ និងទទួលយកព្រះយេស៊ូវគ្រីស្ទដោយស្មោះអស់ពីចិត្ត នឹងទទួលបានសេចក្តីសង្គ្រោះ។ [[ROM 10:13]] [[ROM 10:9-10]] [[REV 3:20]] [[JOH 1:12]]

- ព្រះយេស៊ូវគ្រីស្ទ [[ROM 5:8]]
- ហោរាដ៏ពិត [[JOH 14:6]]
- បូជាចារ្យដ៏ពិត [[ROM 8:2]]
- ស្តេចដ៏ពិត [[1JO 3:8]]

## សូមអធិស្ឋាននិយាយតាមពាក្យបន្ទាប់នេះ ដោយចេញពីចិត្ត៖

ព្រះដ៏ជាព្រះវរបិតា ខ្ញុំជាមនុស្សមានបាប។ រហូតមកដល់ពេលនេះ ខ្ញុំបានរស់នៅតាមតែចិត្តខ្លួនឯងចង់បាន។ ប៉ុន្តែ ពេលនេះ ខ្ញុំជឿលើការពិតថា ព្រះយេស៊ូវគឺជាព្រះគ្រីស្ទដែលបានសុគតនៅលើឈើឆ្កាង ហើយបានរស់ឡើងវិញដើម្បីខ្ញុំ។ ខ្ញុំសូមបើកទ្វារដួងចិត្ត ហើយទទួលយកព្រះយេស៊ូវជាព្រះអង្គសង្គ្រោះរបស់ខ្ញុំ។ សូមយាងចូលមកក្នុងចិត្តខ្ញុំ ហើយដឹកនាំខ្ញុំរហូតអស់កល្បជានិច្ច។ ខ្ញុំសូមអធិស្ឋានក្នុងព្រះនាម ព្រះយេស៊ូវគ្រីស្ទ។ អាម៉ែន។

## 5. អ្នកដែលបានទទួលសេចក្តីសង្គ្រោះ បានទទួលព្រះពរដ៏ធំ

(1) ឥឡូវនេះ អ្នកគឺជាកូនរបស់ព្រះជាម្ចាស់។ [[JOH 1:12]] [[1CO 3:16]]

(2) ព្រះវិញ្ញាណបរិសុទ្ធតែងតែគង់នៅជាមួយ និងដឹកនាំអ្នកជានិច្ច។ [[JOH 14:16-27]]

(3) ព្រះជាម្ចាស់ឆ្លើយតបនឹងសេចក្តីអធិស្ឋានរបស់អ្នក។ [[JOH 14:13-14]] [[ACT 1:8]]

(4) អ្នកអាចបំបាក់អំណាចនៃសេចក្តីងងឹតដោយព្រះចេស្តារបស់ព្រះយេស៊ូវ។ [[MAR 3:15]] [[1JO 3:8]] [[MAT 12:28-29]] [[LUK 10:19]]

(5) ពេលជួបការលំបាក ប្រសិនបើអ្នកអធិស្ឋាន ព្រះជាម្ចាស់នឹងចាត់កងទ័ពទេវតាមកការពារអ្នក។ [[HEB 1:14]] [[REV 8:3-5]] [[PSA 103:20-22]]

(6) ក្នុងនាមជារាស្ត្ររបស់ព្រះ ដែលមានសញ្ជាតិនៅស្ថានសួគ៌ អ្នកអាចរីករាយនឹងព្រះពរនោះ សូម្បីតែនៅលើផែនដីនេះ។ [[PHI 3:20]]

(7) អ្នកបានស្តារនូវសិទ្ធិអំណាចឡើងវិញ ដើម្បីយកឈ្នះ និងគ្រប់គ្រងពិភពលោកដោយដំណឹងល្អ។ [[MAT 28:16-20]]
